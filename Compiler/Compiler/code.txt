int sum(int a, int b) {
	return a + b;
}

int calculate(int c) {
    if (c > 0) {
        return 1;
    }
    else if (c < 0) {
        return -1;
    }
    return 0;
}

void show(int c) {
	cout << "Sum : " << c << endl;
}

int main () {
	int a = 9;
	int b = 11;

	int c = sum(a, b);

	show(c);

    for (int i = 0; i < c; i++) {
        cout << i << endl;
    }

    int d = calculate(c);

    cout << "Result : " << d << endl;

	return 0;
}
