type			→ <int> | <float> | <double> | <char> | <string> | <bool>
literal			→ <INTEGER> | <DECIMAL> | <CHARACTER> | <STRLITERAL> | <true> | <false>
compare_op		→ <EQUALS> | <NOTEQUAL> | <GREATERorEQUAL> | <SMALLERorEQUAL> | <GREATERTHAN> | <SMALLERTHAN>
logical_op		→ <AND_LOGIC> | <OR_LOGIC>
assign_op		→ <ASSIGN> | <ADD_ASSIGN> | <SUB_ASSIGN> | <MUL_ASSIGN> | <DIV_ASSIGN> | <MOD_ASSIGN>
add_sub			→ <ADD> | <SUBTRACT>
mul_div_mod		→ <MULTIPLY> | <DIVIDE> | <MOD>


program			→ headers declarations

headers			→ header headers | header
header			→ <#include> (<LIBRARY> | <HEADER>) | <using> <namespace> <IDENTIFIER> <;>

declarations	→ declaration declarations | ε
declaration		→ define | object | variable | function | void | main

define			→ <#define> <IDENTIFIER> literal

object			→ (<class> | <struct>) <IDENTIFIER> <{> objBody <}> <;>
objBody			→ objBlock objBody | ε
objBlock		→ access <:> members
access			→ <private> | <public> | <protected>
members			→ member members | ε
member			→ variable | function

variable		→ (<const> | <static> | ε) type <IDENTIFIER> ((<ASSIGN> expression | ε) | multi_var) <;>
multi_var		→ <,> <IDENTIFIER> (multi_var | ε)

function		→ (<const> | <static> | ε) <type> <IDENTIFIER> <(> arguments <)> <{> statements return <}>
main			→ <int> <main> <(> <)> <{> statements (return | ε) <}>
void			→ <void> <IDENTIFIER> <(> arguments <)> <{> statements <}>
return			→ <return> expression <;>

arguments		→ argument (<COMMA> arguments| ε)
argument		→ type <IDENTIFIER>

statements		→ statement statements | ε
statement		→ variable | iostream | assignment | increment <;> | decrement <;> | call_statement | loops | if | return

iostream		→ cout | cin
cin				→ <cin> <RIGHTSHIFT> <IDENTIFIER> <;>
cout			→ <cout> ostrings <;>
ostrings		→ ostring (ostrings | ε)
ostring			→ <LEFTSHIFT> expression

assignment		→ <IDENTIFIER> assign_op expression <;>

expression		→ term subterm
term			→ factor subfactor
subterm			→ add_sub term subterm | ε
factor			→ <(> expression <)> | <IDENTIFIER> | literal | increment | decrement | function_call
subfactor		→ mul_div_mod factor subfactor | ε

increment		→ <INCREMENT> <IDENTIFIER> | <IDENTIFIER> <INCREMENT>
decrement		→ <DECREMENT> <IDENTIFIER> | <IDENTIFIER> <DECREMENT>

function_call	→ <IDENTIFIER> <(> (parameters | ε) <)>
parameters		→ expression (<,> parameters | ε)
call_statement	→ function_call <;>

loops			→ for | while | dowhile
for				→ <for> <(> (type | ε) <IDENTIFIER> <ASSIGN> expression <;> conditions <;> update <)> <{> statements <}>
while			→ <while> <(> conditions <)> <{> statements <}>
dowhile			→ <do> <{> statements <}> <while> <(> conditions <)> <;>

if				→ <if> <(> conditions <)> <{> statements <}> elseif else
elseif			→ <else if> <(> conditions <)> <{> statements <}> elseif | ε
else			→ <else> <{> statements <}> | ε

conditions		→ condition (logical_op conditions | ε)
condition		→ <(> conditions <)> | comparison
comparison		→ expression compare_op expression

update			→ increment |  decrement | <IDENTIFIER> assign_op expression
